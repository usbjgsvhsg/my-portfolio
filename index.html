import socket
def scan_target(target, ports):
    # فحص المنافذ النشطة للأمان واختبار الاختراق
    for port in ports:
        s = socket.socket()
        result = s.connect_ex((target, port))
        if result == 0:
            print(f"Port {port} is OPEN")
